1-Click SQL Script สำหรับตั้งค่า Supabase Table

คัดลอกโค้ดนี้วางใน SQL Editor ของ Supabase เพื่อสร้างตาราง `water_warehouse`

-- สร้างตารางพัสดุประปา WATER SUPPLY WAREHOUSE
CREATE TABLE IF NOT EXISTS water_warehouse (
    id UUID PRIMARY KEY DEFAULT gen_random_policy_uuid(),
    code VARCHAR(100) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,
    category VARCHAR(100) NOT NULL DEFAULT 'อุปกรณ์ทั่วไป',
    location VARCHAR(100) DEFAULT 'คลังใหญ่',
    unit VARCHAR(50) DEFAULT 'ชิ้น',
    quantity INT DEFAULT 0,
    warning_threshold INT DEFAULT 10,
    critical_threshold INT DEFAULT 3,
    usage_notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- เปิดสิทธิ์ Row Level Security (RLS)
ALTER TABLE water_warehouse ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow All Access" ON water_warehouse FOR ALL USING (true) WITH CHECK (true);
